
import {copyFileSync,constants as cpFlags,existsSync,writeFileSync,appendFileSync} from 'node:fs'
const qaOwners=new WeakMap(); const qaLoad=NativePreparation.prototype.loadSourceSnapshot
NativePreparation.prototype.loadSourceSnapshot=function(desc){qaOwners.set(this,desc);return qaLoad.call(this,desc)}
const qaCall=NativePreparation.prototype.request
NativePreparation.prototype.request=function(q){
 const d=qaOwners.get(this),out=process.env.QUEUE_HIGH_CAPTURE
 if(d&&out&&q.method==='compile_gpu_packet'&&q.lane==='canvas'){
  const o=JSON.parse(q.args[0])
  if(Math.abs(o.view.zoom-64)<0.0001&&Math.abs(o.view.x+252366.8162)<0.1&&Math.abs(o.view.y-117504.0081)<0.1){
   const name=path.join(out,'source.scene'),file=path.join(out,'store.json')
   if(!existsSync(file)){copyFileSync(d.path,name,cpFlags.COPYFILE_FICLONE);writeFileSync(file,JSON.stringify({...d,path:name}))}
   appendFileSync(path.join(out,'requests.ndjson'),JSON.stringify(o)+'\n')
  }
 }
 return qaCall.call(this,q)
}
