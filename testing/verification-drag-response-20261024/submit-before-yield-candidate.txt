  private yieldEncoding() {
    this.checkFrameCurrent()
    const pause = this.encodeYield.checkpoint(this.driver.captureDisplay === true || !!this.frameCurrent)
    const submitBeforeYield=import.meta.env.VITE_DESIGN_SUBMIT_BEFORE_YIELD==='1'&&import.meta.env.VITE_DESIGN_ENCODE_SUBMIT!=='0'
    // Candidate for macOS Dawn/IOSurface scheduling waits: flush the already
    // encoded producer work before yielding to a canvas presentation task.
    // Submitted resources retain the existing completion/abort ownership.
    if(pause&&submitBeforeYield)this.driver.submitPending?.()
    return pause?.then(() => {
      // Let control messages invalidate this frame before its next GPU submit.
      // Cancellation abandons this unsubmitted encoder through the existing
      // abort/fence path; already submitted resources keep that same lifetime.
      this.checkFrameCurrent()
      if (!submitBeforeYield&&import.meta.env.VITE_DESIGN_ENCODE_SUBMIT !== '0') this.driver.submitPending?.()
    })
  }
