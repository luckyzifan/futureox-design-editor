import { WorkspaceAutomationHost } from '__DESIGN_REPO__/frontend/design-desktop/src/workspace-automation.mjs'
it('paginates all 1001 real SQLite revisions through bounded facade pages',async()=>{
 const dir=await mkdtemp(path.join(tmpdir(),'independent-history-'));close.push(()=>rm(dir,{recursive:true,force:true}));
 const sidecar=new Sidecar('__BUILD_ROOT__/cargo-target/debug/futureox-design-desktop-backend',path.join(dir,'recovery'),{workspaceDb:path.join(dir,'workspace.sqlite3'),catalogDb:path.join(dir,'catalog.sqlite3')});close.push(()=>sidecar.stop());
 const document=blank('Long edit history');
 for(let i=0;i<1001;i++){document.name='Revision '+i;await sidecar.request('workspace.commit',{document,expectedRevision:i||null,mutationId:'history_'+i})}
 const raw=await sidecar.request('workspace.revisions',{id:document.id});console.log('RAW_HISTORY_PROBE',raw.revisions.length);
 const root={};const state={documentId:document.id,session:0,revision:0,storageRevision:1001,dirty:false,busy:false,pendingEdits:false,interactionActive:false};
 const host=new WorkspaceAutomationHost({getRoot:()=>root,status:()=>({displayName:'temp'}),storage:(m:any,p:any)=>sidecar.request(m,p),dispatch:async()=>state});
 const grant=await host.grant('reviewer',{workspace:'read'});const handle=host.issue(grant,'document',document.id,document.id);
 const out=await host.execute({sessionId:'reviewer',documentId:document.id,session:0,deadline:Date.now()+10000,request:{method:'workspace.revisions',documentId:document.id,session:0,params:{rootHandle:grant.rootHandle,documentHandle:handle}}},new AbortController().signal);
 expect(out.result.revisions.length).toBeLessThanOrEqual(100);expect(out.result.revisions[0].storageRevision).toBe(1001);expect(out.result.hasMore).toBe(true); const seen=out.result.revisions.map((r:any)=>r.storageRevision);let next=out.result;while(next.hasMore){const page=await host.execute({sessionId:'reviewer',documentId:document.id,session:0,deadline:Date.now()+10000,request:{method:'workspace.revisions',documentId:document.id,session:0,params:{rootHandle:grant.rootHandle,documentHandle:handle,beforeRevision:next.nextBeforeRevision}}},new AbortController().signal);next=page.result;expect(next.revisions.length).toBeLessThanOrEqual(100);seen.push(...next.revisions.map((r:any)=>r.storageRevision))};expect(seen).toEqual(Array.from({length:1001},(_,i)=>1001-i))
})
