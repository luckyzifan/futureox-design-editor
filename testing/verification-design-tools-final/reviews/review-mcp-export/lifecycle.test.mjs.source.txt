import test from 'node:test'
import assert from 'node:assert/strict'
import { PassThrough } from 'node:stream'
import { DesignMcpServer } from '__DESIGN_REPO__/frontend/design-mcp/src/server.mjs'
import { ArtifactResources } from '__DESIGN_REPO__/frontend/design-mcp/src/artifacts.mjs'
const preview = data => ({ok:true,method:'preview.capture',revision:7,result:{format:'futureox.design-preview.v1',mimeType:'image/png',dataUrl:`data:image/png;base64,${data}`}})
const deferred = () => {let resolve;const promise=new Promise(r=>resolve=r);return {promise,resolve}}
const turn=()=>new Promise(r=>setImmediate(r))
function setup(callTool){
 const streams=[new PassThrough(),new PassThrough(),new PassThrough()];for(const s of streams)s.resume()
 const server=new DesignMcpServer({backend:{listTools:()=>[{name:'design_preview_capture',description:'Actual preview',inputSchema:{type:'object'}}],callTool},input:streams[0],output:streams[1],diagnostics:streams[2]})
 return {server,cleanup:async()=>{await server.close();streams.forEach(s=>s.destroy())}}
}
test('cancelled asynchronous exports do not publish artifacts or evict prior exact bytes',async t=>{
 const gate=deferred(), f=setup(()=>gate.promise);t.after(f.cleanup)
 f.server.artifacts=new ArtifactResources({maxArtifacts:1})
 const prior=f.server.artifacts.project(preview('AQ==')),uri=prior.links[0].uri, before=f.server.artifacts.read(uri)
 const controller=new AbortController(), work=f.server.callTool({name:'design_preview_capture'},{signal:controller.signal})
 await turn();controller.abort(new Error('Client explicitly cancelled'));gate.resolve(preview('Ag=='))
 await assert.rejects(work,/Client explicitly cancelled/)
 assert.deepEqual(f.server.artifacts.read(uri),before);assert.equal(f.server.artifacts.list().length,1)
})
test('late asynchronous export after close cannot repopulate cleared artifacts even without caller abort',async t=>{
 const gate=deferred(),f=setup(()=>gate.promise);t.after(f.cleanup)
 f.server.artifacts.project(preview('AQ=='))
 const work=f.server.callTool({name:'design_preview_capture'},{signal:new AbortController().signal})
 await turn();await f.server.close();assert.equal(f.server.artifacts.bytes,0)
 gate.resolve(preview('Ag=='));await assert.rejects(work,/MCP session closed/)
 assert.equal(f.server.artifacts.list().length,0);assert.equal(f.server.artifacts.bytes,0)
})
test('normal method failure remains a tool result rather than transport exception',async t=>{
 const f=setup(async()=>{throw Object.assign(new Error('Expected renderer unavailable'),{code:'renderer_unavailable'})});t.after(f.cleanup)
 const result=await f.server.callTool({name:'design_preview_capture'},{signal:new AbortController().signal})
 assert.equal(result.isError,true);assert.equal(result.structuredContent.error.code,'renderer_unavailable');assert.equal(f.server.artifacts.list().length,0)
})
test('successful exact preview bytes remain a resource and native image when under tool budget',async t=>{
 const f=setup(async()=>preview('AQIDBA=='));t.after(f.cleanup)
 const result=await f.server.callTool({name:'design_preview_capture'},{signal:new AbortController().signal})
 assert.equal(result.isError,false);assert.equal(result.structuredContent.result.dataUrl,undefined)
 const image=result.content.find(c=>c.type==='image'),link=result.content.find(c=>c.type==='resource_link')
 assert.equal(image.data,'AQIDBA==');assert.equal(f.server.artifacts.read(link.uri).contents[0].blob,image.data)
 assert(Buffer.byteLength(JSON.stringify(result))<=256*1024-2048)
})
