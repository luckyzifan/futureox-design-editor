import assert from 'node:assert/strict'
import { EventEmitter } from 'node:events'
import { spawnSync } from 'node:child_process'

import { NativeRenderFrame, MAX_METADATA_BYTES } from '__DESIGN_REPO__/frontend/design-desktop/src/native-render-frame.mjs'
import { MAX_FRAME_BYTES } from '__DESIGN_REPO__/frontend/design-desktop/src/security.mjs'
const binary='__BUILD_ROOT__/cargo-target/debug/design-render'
function fixture(t,protocol='frb1'){
 const stream=()=>new EventEmitter(),child=Object.assign(new EventEmitter(),{stdin:stream(),stdout:stream(),stderr:stream(),exitCode:null,signalCode:null});let writes=0
 child.stdin.write=()=>{writes++;return true};child.kill=signal=>{queueMicrotask(()=>{child.signalCode=signal;child.emit('close')});return true}
 const bridge=new NativeRenderProcess('test',undefined,'cpu',{protocol,spawnProcess:()=>child});bridge.spawnProcess();t.after(()=>bridge.stop());return {bridge,child,get writes(){return writes}}
}
function requestWithSize(bridge,bytes){const base={cmd:'stats',padding:''};const extra=bridge.protocol==='frb1'?{session:bridge.sessionId,id:bridge.nextId}:{};return {...base,padding:' '.repeat(bytes-Buffer.byteLength(JSON.stringify({...base,...extra}))-1)}}
function envelope(metadata,pixels=Buffer.alloc(0)){const m=Buffer.isBuffer(metadata)?metadata:Buffer.from(JSON.stringify(metadata));const h=Buffer.alloc(12);h.write('FRB1');h.writeUInt32LE(m.length,4);h.writeUInt32LE(pixels.length,8);return Buffer.concat([h,m,pixels])}
const expected={session:'audit',id:1,kind:'stats'}
function decode(wire){const f=new NativeRenderFrame();f.append(wire);return f.decode(expected)}
function command(message){return JSON.stringify(message)+'\n'}
function run(input){const r=spawnSync(binary,['--serve','--backend','cpu','--serve-protocol','frb1'],{input,maxBuffer:8*1024*1024,timeout:20000});assert.equal(r.error,undefined);return r}
function responses(bytes){const result=[];let pos=0;while(pos<bytes.length){assert.equal(bytes.subarray(pos,pos+4).toString(),'FRB1');const n=bytes.readUInt32LE(pos+4),p=bytes.readUInt32LE(pos+8);result.push(JSON.parse(bytes.subarray(pos+12,pos+12+n)));pos+=12+n+p}return result}
for(const protocol of ['json','frb1']){
 test(`${protocol}: request body exact 64 MiB accepted, one byte over rejected, accounting drains`,async t=>{
  const f=fixture(t,protocol),body=requestWithSize(f.bridge,MAX_FRAME_BYTES+1),pending=assert.rejects(f.bridge.request('stats',body),/Bridge stopped/)
  assert.equal(f.bridge.pendingBytes,MAX_FRAME_BYTES+1);assert.equal(f.writes,1)
  await assert.rejects(f.bridge.request('stats',requestWithSize(f.bridge,MAX_FRAME_BYTES+2)),/budget/)
  assert.equal(f.bridge.pendingBytes,MAX_FRAME_BYTES+1);await f.bridge.stop();await pending;assert.equal(f.bridge.pendingBytes,0)
 })
 test(`${protocol}: aggregate request budget accepts exact 96 MiB and rejects the next byte-bearing request`,async t=>{
  const f=fixture(t,protocol),a=assert.rejects(f.bridge.request('stats',requestWithSize(f.bridge,48*1024*1024)),/Bridge stopped/),b=assert.rejects(f.bridge.request('stats',requestWithSize(f.bridge,48*1024*1024)),/Bridge stopped/)
  assert.equal(f.bridge.pendingBytes,NATIVE_RENDER_MAX_PENDING_BYTES);assert.equal(f.writes,1);await assert.rejects(f.bridge.stats(),/budget/);assert.equal(f.bridge.queue.length,1)
  await f.bridge.stop();await Promise.all([a,b]);assert.equal(f.bridge.pendingBytes,0)
 })
}
test('safe request id limit sends last safe id once, then rejects without writing',async t=>{
 const f=fixture(t);f.bridge.nextId=Number.MAX_SAFE_INTEGER;const pending=assert.rejects(f.bridge.stats(),/Bridge stopped/);assert.equal(f.bridge.pending.id,Number.MAX_SAFE_INTEGER);await assert.rejects(f.bridge.stats(),/sequence exhausted/);assert.equal(f.writes,1);await f.bridge.stop();await pending
})
test('metadata accepts exact 64 KiB, rejects one byte over',()=>{
 const meta={session:'audit',id:1,type:'stats',stats:{padding:''}},overhead=Buffer.byteLength(JSON.stringify(meta));meta.stats.padding=' '.repeat(MAX_METADATA_BYTES-overhead);assert.equal(decode(envelope(meta)).stats.padding.length,MAX_METADATA_BYTES-overhead);meta.stats.padding+=' ';assert.throws(()=>decode(envelope(meta)),/limit/)
})
test('all high-bit magic corruptions must fail closed',()=>{
 for(let index=0;index<4;index++){const wire=envelope({session:'audit',id:1,type:'stats',stats:{}});wire[index]|=128;assert.throws(()=>decode(wire),/protocol/,`magic byte ${index} high bit must be rejected`)}
})
test('invalid UTF-8 metadata must fail closed rather than replacing bytes',()=>{
 const json=Buffer.from(JSON.stringify({session:'audit',id:1,type:'stats',stats:{label:'x'}}));json[json.indexOf(Buffer.from('"x"'))+1]=0xff;assert.throws(()=>decode(envelope(json)),/UTF|encoding|encoded|valid/i)
})
test('actual Rust process validates missing, unsafe and malformed identities',()=>{
 for(const identity of [{},{session:'audit'},{session:'audit',id:0},{session:'audit',id:-1},{session:'audit',id:1.5},{session:'audit',id:Number.MAX_SAFE_INTEGER+1},{session:'',id:1},{session:'a'.repeat(65),id:1},{session:'中',id:1},{session:'invalid token',id:1}]){const r=run(command({cmd:'stats',...identity}));assert.notEqual(r.status,0,JSON.stringify(identity));assert.equal(r.stdout.length,0)}
 const valid=run(command({cmd:'stats',session:'a'.repeat(64),id:Number.MAX_SAFE_INTEGER}));assert.equal(valid.status,0);assert.equal(responses(valid.stdout)[0].id,Number.MAX_SAFE_INTEGER)
})
test('actual Rust process rejects replay, decreasing id and foreign session',()=>{
 for(const second of [{session:'audit',id:1},{session:'audit',id:0},{session:'other',id:2}]){const r=run(command({cmd:'stats',session:'audit',id:1})+command({cmd:'stats',...second}));assert.notEqual(r.status,0);assert.equal(responses(r.stdout).length,1)}
})
test('actual Rust command errors remain scoped and next command proceeds',()=>{
 const r=run(command({cmd:'unknown',session:'audit',id:1})+command({cmd:'stats',session:'audit',id:2}));assert.equal(r.status,0);assert.deepEqual(responses(r.stdout).map(x=>[x.type,x.id,x.session]),[['error',1,'audit'],['error',2,'audit']])
})
test('actual Rust rejects truncated, invalid JSON and invalid UTF-8 commands',()=>{
 for(const bytes of [Buffer.from(JSON.stringify({cmd:'stats',session:'audit',id:1})),Buffer.from('{"\n'),Buffer.from([0xff,10])]){const r=run(bytes);assert.notEqual(r.status,0);assert.equal(r.stdout.length,0)}
})
test('actual Rust accepts exact 64 MiB command body and rejects one byte over',()=>{
 const base={cmd:'stats',session:'audit',id:1,padding:''},overhead=Buffer.byteLength(JSON.stringify(base));base.padding=' '.repeat(MAX_FRAME_BYTES-overhead);const good=run(command(base));assert.equal(good.status,0);assert.equal(responses(good.stdout)[0].id,1);base.padding+=' ';const bad=run(command(base));assert.notEqual(bad.status,0);assert.equal(bad.stdout.length,0);assert.match(bad.stderr.toString(),/exceeds limit/)
