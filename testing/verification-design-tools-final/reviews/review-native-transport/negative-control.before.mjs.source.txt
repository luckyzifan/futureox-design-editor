import {readFileSync,writeFileSync,mkdtempSync,rmSync,cpSync} from 'node:fs'
import {tmpdir} from 'node:os'
import path from 'node:path'
import {spawnSync} from 'node:child_process'
const root=new URL('../../',import.meta.url).pathname,temp=mkdtempSync(path.join(tmpdir(),'native-render-corruption-'))
try{
 cpSync(path.join(root,'frontend/design-desktop/src/native-render-frame.mjs'),path.join(temp,'frame.mjs'))
 let source=readFileSync(path.join(temp,'frame.mjs'),'utf8');source=source.replace('result.rgba = rgba','rgba[rgba.length - 1] ^= 1; result.rgba = rgba');writeFileSync(path.join(temp,'frame.mjs'),source)
 let test=readFileSync(path.join(root,'frontend/design-desktop/test/native-render-frame.test.mjs'),'utf8').replaceAll('../src/native-render-frame.mjs','./frame.mjs');writeFileSync(path.join(temp,'frame.test.mjs'),test)
 const result=spawnSync(process.execPath,['--test',path.join(temp,'frame.test.mjs')],{encoding:'utf8'})
 process.stdout.write(result.stdout+result.stderr)
 if(result.status===0||!result.stdout.includes('fail'))throw Error('Corrupted payload was not caught')
 console.log('PASS: production payload corruption was detected by the byte-preservation tests')
}finally{rmSync(temp,{recursive:true,force:true})}
