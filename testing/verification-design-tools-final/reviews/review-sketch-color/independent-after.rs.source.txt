use futureox_design_kernel::{color::{self,ColorSpace},interop::{export_sketch,import_sketch,read_zip_public},scene::Document};
use serde_json::{Value,json};
fn doc(color:&str,space:Option<&str>)->Document {let mut d=json!({"schemaVersion":13,"id":"d","name":"d","pages":[{"id":"p","name":"p","children":["n"],"background":"#ffffff"}],"nodes":{"n":{"id":"n","kind":"rectangle","width":10,"height":10,"fills":[{"kind":"solid","color":color}]}}}); if let Some(s)=space {d["colorManagement"]=json!({"documentSpace":s,"referenceWhiteNits":203.,"peakLuminanceNits":1000.});}serde_json::from_value(d).unwrap()}
fn parts(d:&Document)->(Value,Value){parts_bytes(&export_sketch(d).unwrap())}
fn parts_bytes(b:&[u8])->(Value,Value){let f=read_zip_public(b).unwrap();(serde_json::from_slice(&f["document.json"]).unwrap(),serde_json::from_slice(f.iter().find(|(k,_)|k.starts_with("pages/")).unwrap().1).unwrap())}
fn rgba(c:&Value)->[f64;4]{["red","green","blue","alpha"].map(|k|c[k].as_f64().unwrap())}
fn fill(d:&Document)->Value{parts(d).1["layers"][0]["style"]["fills"][0].clone()}
fn error(d:&Document){assert!(export_sketch(d).is_err());}
