let dom: JSDOM, mount: HTMLDivElement, root: ReturnType<typeof createRoot>
beforeEach(() => {
  dom = new JSDOM('<!doctype html><body></body>', { url: 'https://design.invalid/' })
  for (const name of ['window','document','navigator','HTMLElement','Node','Event','KeyboardEvent','localStorage']) vi.stubGlobal(name, name === 'window' ? dom.window : dom.window[name])
  vi.stubGlobal('IS_REACT_ACT_ENVIRONMENT', true)
  settingsPreference.reset()
  mount = document.createElement('div'); document.body.append(mount); root = createRoot(mount)
})
afterEach(async () => { await act(async () => root.unmount()); dom.window.close(); vi.unstubAllGlobals() })

async function fixture(mutate = false) {
  const listeners = new Set<() => void>()
  const editor = { state: { scrollWheelZoom: false, snap: true, snapPixels: true, pixelGrid: true }, subscribe: (fn: () => void) => { listeners.add(fn); return () => listeners.delete(fn) }, snapshot: () => editor.state, notify: vi.fn() }
  let setSettings: (value: boolean) => void, setPrompt: (value: boolean) => void, incoming: (value: { requestId: string }) => void
  function Shell() {
    const [settings, updateSettings] = useState(true), [prompt, updatePrompt] = useState(false)
    setSettings = updateSettings; setPrompt = updatePrompt
    return React.createElement(React.Fragment, null,
      React.createElement('button', { 'data-testid': 'canvas' }, 'Canvas'),
      prompt && React.createElement('section', { role: 'alertdialog', 'aria-modal': true }, React.createElement('button', { autoFocus: true, 'data-dirty': true }, 'Cancel close')),
      settings && React.createElement(SettingsDialog, { editor, onClose: () => updateSettings(false) }))
  }
  await act(async () => root.render(React.createElement(Shell)))
  const desktopSession = { closeReady: false, requestClose: vi.fn(async () => { setPrompt(true); return false }), cancelClose: vi.fn() }
  const desktop = { onCloseRequested: (handler: typeof incoming) => { incoming = handler }, respondToClose: vi.fn(async () => ({ accepted: true })) }
  const app = readFileSync('<repo>/frontend/design-client/src/App.tsx', 'utf8')
  let callback = app.slice(app.indexOf('const offClose = desktop?.onCloseRequested'), app.indexOf('    return () => { window.removeEventListener', app.indexOf('const offClose = desktop?.onCloseRequested')))
  expect(callback).toContain('desktopSession.requestClose()')
  if (mutate) callback = callback.replace(/flushSync\(\(\) => setShowSettings\(false\)\)/, 'void 0')
  Function('desktop','desktopSession','fileImports','ready','editor','flushSync','setShowSettings', callback)(desktop, desktopSession, { invalidate() {} }, true, editor, flushSync, setSettings!)
  await act(async () => incoming!({ requestId: 'native-close' }))
  return { desktop, desktopSession }
}

test('actual App close callback releases Settings portal/inert before dirty confirmation focuses', async () => {
  const f = await fixture()
  expect(document.querySelector('.settings-dialog')).toBeNull()
  expect(mount.hasAttribute('inert')).toBe(false)
  expect(document.activeElement).toBe(document.querySelector('[data-dirty]'))
  expect(f.desktop.respondToClose).toHaveBeenCalledWith({ allow: false, requestId: 'native-close' })
  expect(f.desktopSession.cancelClose).toHaveBeenCalledOnce()
})

test('negative control: remove App close cleanup and dirty confirmation becomes inert and unfocusable', async () => {
  await fixture(true)
  expect(document.querySelector('.settings-dialog')).not.toBeNull()
  expect(mount.hasAttribute('inert')).toBe(true)
  expect(document.activeElement).toBe(document.querySelector('[aria-label="关闭设置"]'))
  expect(document.activeElement).not.toBe(document.querySelector('[data-dirty]'))
})

import { WorkspaceRootControls } from '<repo>/frontend/design-client/src/WorkspaceRootControls'
const labelButton = (label: string) => [...mount.querySelectorAll<HTMLButtonElement>('button')].find(value => value.textContent === label)!
const rootStatus = (canSwitch = true, isDefault = false) => ({ status: 'ok', displayName: 'Fixture workspace', canSwitch, isDefault })
const deferred = () => { let resolve!: (value: any) => void, reject!: (value: any) => void; const promise = new Promise((a, b) => { resolve = a; reject = b }); return { promise, resolve, reject } }
function rootBridge(status: any) { return { status, select: vi.fn(), useDefault: vi.fn(), activate: vi.fn() } }

test('peer closes while Settings stays open: explicit refresh reenables current buttons and preserves default guard', async () => {
  const bridge = rootBridge(vi.fn().mockResolvedValue(rootStatus(false, true))), choose = vi.fn()
  await act(async () => root.render(React.createElement(WorkspaceRootControls, { bridge, busy: false, onChoose: choose })))
  expect(labelButton('切换本地工作区…').disabled).toBe(true)
  bridge.status.mockResolvedValue(rootStatus(true, true))
  await act(async () => labelButton('重新检查窗口').click())
  expect(bridge.status).toHaveBeenCalledTimes(2)
  expect(labelButton('切换本地工作区…').disabled).toBe(false)
  expect(labelButton('打开原默认工作区').disabled).toBe(true)
  await act(async () => labelButton('切换本地工作区…').click())
  expect(choose).toHaveBeenCalledExactlyOnceWith(false)
})

test('status rejection fails closed and refresh recovers without selecting or activating', async () => {
  const bridge = rootBridge(vi.fn().mockRejectedValue(new Error('Disconnected')))
  await act(async () => root.render(React.createElement(WorkspaceRootControls, { bridge, busy: false, onChoose: vi.fn() })))
  expect(mount.querySelector('[role="alert"]')?.textContent).toContain('Disconnected')
  expect(labelButton('切换本地工作区…').disabled).toBe(true)
  bridge.status.mockResolvedValue(rootStatus())
  await act(async () => labelButton('重新检查窗口').click())
  expect(mount.querySelector('[role="alert"]')).toBeNull()
  expect(labelButton('切换本地工作区…').disabled).toBe(false)
  expect(labelButton('打开原默认工作区').disabled).toBe(false)
  expect(bridge.select).not.toHaveBeenCalled(); expect(bridge.activate).not.toHaveBeenCalled()
})

test('refresh in flight disables native choices; old bridge results cannot overwrite current status', async () => {
  const pending = deferred(), old = rootBridge(vi.fn().mockReturnValue(pending.promise)), current = rootBridge(vi.fn().mockResolvedValue(rootStatus(false)))
  await act(async () => root.render(React.createElement(WorkspaceRootControls, { bridge: old, busy: false, onChoose: vi.fn() })))
  expect(labelButton('切换本地工作区…').disabled).toBe(true)
  expect(labelButton('重新检查窗口').disabled).toBe(true)
  await act(async () => root.render(React.createElement(WorkspaceRootControls, { bridge: current, busy: false, onChoose: vi.fn() })))
  await act(async () => pending.resolve(rootStatus(true)))
  expect(labelButton('切换本地工作区…').disabled).toBe(true)
  expect(labelButton('重新检查窗口').disabled).toBe(false)
})

test('root status settling after unmount performs no UI work', async () => {
  const pending = deferred(), bridge = rootBridge(vi.fn().mockReturnValue(pending.promise)), error = vi.spyOn(console, 'error')
  await act(async () => root.render(React.createElement(WorkspaceRootControls, { bridge, busy: false, onChoose: vi.fn() })))
  await act(async () => root.render(null))
  await act(async () => pending.reject(new Error('Late transport failure')))
  expect(mount.innerHTML).toBe(''); expect(error).not.toHaveBeenCalled(); error.mockRestore()
})
