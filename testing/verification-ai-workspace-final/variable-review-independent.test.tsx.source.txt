function editor(doc = blank()) { const e = new Editor(doc); editors.push(e); return e }
function fixture(count = 2, modeCount = 2) {
  const doc = blank('Variable UI'), layer = node('rectangle', 0, 0, 10, 10), s = doc.designSystem = emptySystem()
  doc.schemaVersion = 4
  doc.nodes[layer.id] = layer; doc.pages[0].children = [layer.id]
  const modes = ['Light', 'Dark', ...Array.from({ length: Math.max(0, modeCount - 2) }, (_, i) => `Mode ${i + 3}`)]
  s.collections.theme = { name: 'Theme', modes, defaultMode: 'Light' }; s.activeModes.theme = 'Light'
  for (let i = 0; i < count; i++) s.variables[`v${i}`] = { name: `Spacing/${String(i).padStart(4, '0')}`, collection: 'theme', kind: 'number', values: Object.fromEntries(modes.map((m, j) => [m, 10 + i + j * 70])) }
  return { doc, layer }
}
const label = <T extends HTMLElement = HTMLInputElement>(name: string): T => {
  const found = [...mount.querySelectorAll<HTMLElement>('[aria-label]')].find(e => e.getAttribute('aria-label') === name)
  expect(found, `Missing label ${name}`).toBeTruthy(); return found as T
}
const button = (name: string) => {
  const found = [...mount.querySelectorAll<HTMLButtonElement>('button')].find(b => b.textContent?.trim() === name)
  expect(found, `Missing button ${name}`).toBeTruthy(); return found!
}
function click(element: HTMLElement) { act(() => element.click()) }
function type(field: HTMLInputElement, value: string) {
  act(() => {
    field.focus(); Object.getOwnPropertyDescriptor(dom.window.HTMLInputElement.prototype, 'value')!.set!.call(field, value)
    field.dispatchEvent(new dom.window.Event('input', { bubbles: true }))
  })
}
function key(field: HTMLElement, value: string) { act(() => field.dispatchEvent(new dom.window.KeyboardEvent('keydown', { key: value, bubbles: true }))) }
function choose(name: string, value: string) {
  const field = label<HTMLSelectElement>(name)
  act(() => { field.value = value; field.dispatchEvent(new dom.window.Event('change', { bubbles: true })) })
}
function finish(name: string, value: string) { const field = label(name); type(field, value); key(field, 'Enter') }

test('invalid variable draft vetoes actual editor document replacement, then Escape permits load', async () => {
  const { doc } = fixture(), e = editor(doc)
  await act(async () => root.render(<DesignTools editor={e} onClose={() => {}} />))
  const before = e.state.doc, epoch = e.sessionGeneration, replacement = blank('Replacement')
  const field = label('Spacing/0000 Light'); type(field, 'bad')
  act(() => { expect(e.load(replacement)).toBe(false) })
  expect(e.state.doc).toBe(before); expect(e.sessionGeneration).toBe(epoch)
  expect(label('Spacing/0000 Light')).toBe(field); expect(field.value).toBe('bad')
  key(field, 'Escape')
  act(() => { expect(e.load(replacement)).toBe(true) })
  expect(e.state.doc.id).toBe(replacement.id); expect(hasPendingEdits(e)).toBe(false)
})

test.each(['helper', 'automation'] as const)('mode rename via %s preserves inherited children and exact explicit overrides through later template edits', async route => {
  const { doc, layer } = fixture(1); doc.schemaVersion = 6
  const { insertInstance, stampKeys } = await import('<repo>/frontend/design-client/src/components')
  const { originNode } = await import('<repo>/frontend/design-client/src/component-fixtures')
  const component = node('frame', 0, 0, 200, 100)
  component.component = { variants: {}, properties: {} }; component.children = [layer.id]
  layer.variableModes = { theme: 'Dark' }; layer.bindings = { width: 'v0' }
  doc.nodes[component.id] = component; doc.pages[0].children = [component.id]
  stampKeys(doc, component.id)
  const inherited = insertInstance(doc, component.id, doc.pages[0].id, { x: 250, y: 0 })
  const explicit = insertInstance(doc, component.id, doc.pages[0].id, { x: 500, y: 0 })
  const childKey = layer.componentKey!
  doc.nodes[inherited].instance!.overrides.$root = { name: 'Unrelated inherited instance label' }
  doc.nodes[explicit].instance!.overrides[childKey] = { variableModes: { theme: 'Dark' }, opacity: 0.45, name: 'Explicit child label' }
  const e = editor(doc), before = e.state.doc
  const getInherited = () => originNode(e.state.doc, inherited, childKey)
  const getExplicit = () => originNode(e.state.doc, explicit, childKey)
  const beforeInheritedOverrides = structuredClone(e.state.doc.nodes[inherited].instance!.overrides)
  const expectedExplicitOverrides = structuredClone(e.state.doc.nodes[explicit].instance!.overrides)
  expectedExplicitOverrides[childKey].variableModes = { theme: 'Night' }
  expect(getInherited().width).toBe(80); expect(getExplicit().width).toBe(80)
  if (route === 'helper') act(() => e.change('Rename mode', d => editing.renameVariableMode(d, 'theme', 'Dark', 'Night')))
  else {
    const { createDesignAutomation } = await import('<repo>/frontend/design-client/src/automation/core')
    const { designSystemExtensions, validateDesignSystemDocument } = await import('<repo>/frontend/design-client/src/automation/design-system')
    const api = createDesignAutomation(e, designSystemExtensions, [validateDesignSystemDocument])
    act(() => {
      const result = api.execute({ method: 'collection.renameMode', params: { collectionId: 'theme', mode: 'Dark', name: 'Night' }, documentId: doc.id, session: e.sessionGeneration, expectedRevision: e.state.version })
      expect(result.ok, JSON.stringify(result)).toBe(true)
    })
  }
  expect(e.state.doc.designSystem!.collections.theme.modes, e.state.message).toEqual(['Light', 'Night'])
  expect(e.state.doc.nodes[inherited].instance!.overrides).toEqual(beforeInheritedOverrides)
  expect(e.state.doc.nodes[explicit].instance!.overrides).toEqual(expectedExplicitOverrides)
  expect(getInherited().variableModes).toEqual({ theme: 'Night' }); expect(getInherited().width).toBe(80)
  expect(getExplicit().variableModes).toEqual({ theme: 'Night' }); expect(getExplicit().width).toBe(80)
  act(() => e.change('Later template mode edit', d => { d.nodes[layer.id].variableModes = { theme: 'Light' } }))
  expect(getInherited().variableModes).toEqual({ theme: 'Light' }); expect(getInherited().width).toBe(10)
  expect(getExplicit().variableModes).toEqual({ theme: 'Night' }); expect(getExplicit().width).toBe(80)
  expect(e.state.doc.nodes[inherited].instance!.overrides).toEqual(beforeInheritedOverrides)
  expect(e.state.doc.nodes[explicit].instance!.overrides).toEqual(expectedExplicitOverrides)
  const after = e.state.doc
  act(() => { e.undo(); e.undo() }); expect(e.state.doc).toEqual(before)
  act(() => { e.redo(); e.redo() }); expect(e.state.doc).toEqual(after)
  const reopened = editor(decodeDocumentBinary(encodeDocumentBinary(after)) as DesignDocument)
  act(() => reopened.change('Later reopened template edit', d => { d.nodes[layer.id].variableModes = { theme: 'Night' } }))
  expect(originNode(reopened.state.doc, inherited, childKey).width).toBe(80)
  expect(reopened.state.doc.nodes[inherited].instance!.overrides).toEqual(beforeInheritedOverrides)
  expect(reopened.state.doc.nodes[explicit].instance!.overrides).toEqual(expectedExplicitOverrides)
})

test('intentional generated-child mode edit in the rename transaction still authors a real override', async () => {
  const { doc, layer } = fixture(1); doc.schemaVersion = 6
  const { insertInstance, stampKeys } = await import('<repo>/frontend/design-client/src/components')
  const { originNode } = await import('<repo>/frontend/design-client/src/component-fixtures')
  const component = node('frame', 0, 0, 200, 100)
  component.component = { variants: {}, properties: {} }; component.children = [layer.id]
  layer.variableModes = { theme: 'Dark' }; layer.bindings = { width: 'v0' }
  doc.nodes[component.id] = component; doc.pages[0].children = [component.id]; stampKeys(doc, component.id)
  const owner = insertInstance(doc, component.id, doc.pages[0].id, { x: 250, y: 0 }), childKey = layer.componentKey!
  const e = editor(doc), generatedId = originNode(e.state.doc, owner, childKey).id
  act(() => e.change('Rename and intentionally override child', d => {
    editing.renameVariableMode(d, 'theme', 'Dark', 'Night')
    d.nodes[generatedId].variableModes = { theme: 'Light' }
  }))
  expect(e.state.doc.designSystem!.collections.theme.modes, e.state.message).toEqual(['Light', 'Night'])
  expect(e.state.doc.nodes[owner].instance!.overrides[childKey].variableModes).toEqual({ theme: 'Light' })
  expect(originNode(e.state.doc, owner, childKey).width).toBe(10)
  act(() => e.change('Change template values after override', d => { editing.setVariableCell(d, 'v0', 'Night', 95); d.nodes[layer.id].variableModes = { theme: 'Night' } }))
  expect(e.state.doc.nodes[layer.id].width).toBe(95)
  expect(originNode(e.state.doc, owner, childKey).width).toBe(10)
})

test.each(['two-collections', 'two-modes', 'value-edit', 'name-swap'] as const)('actual automation batch %s preserves inheritance or explicitly rejects atomically', async variant => {
  const { doc, layer } = fixture(1); doc.schemaVersion = 6
  const { insertInstance, stampKeys } = await import('<repo>/frontend/design-client/src/components')
  const { originNode } = await import('<repo>/frontend/design-client/src/component-fixtures')
  const { createDesignAutomation } = await import('<repo>/frontend/design-client/src/automation/core')
  const { designSystemExtensions, validateDesignSystemDocument } = await import('<repo>/frontend/design-client/src/automation/design-system')
  doc.designSystem!.collections.other = { name: 'Other', modes: ['Light', 'Dark'], defaultMode: 'Light' }
  const component = node('frame', 0, 0, 200, 100)
  component.component = { variants: {}, properties: {} }; component.children = [layer.id]
  layer.variableModes = { theme: 'Dark' }; layer.bindings = { width: 'v0' }
  doc.nodes[component.id] = component; doc.pages[0].children = [component.id]; stampKeys(doc, component.id)
  const owner = insertInstance(doc, component.id, doc.pages[0].id, { x: 250, y: 0 }), childKey = layer.componentKey!, e = editor(doc)
  const before = e.state.doc, version = e.state.version, api = createDesignAutomation(e, designSystemExtensions, [validateDesignSystemDocument])
  const operations = variant === 'name-swap' ? [
    { method: 'collection.renameMode', params: { collectionId: 'theme', mode: 'Dark', name: 'Temp' } },
    { method: 'collection.renameMode', params: { collectionId: 'theme', mode: 'Light', name: 'Dark' } },
    { method: 'collection.renameMode', params: { collectionId: 'theme', mode: 'Temp', name: 'Light' } },
  ] : [{ method: 'collection.renameMode', params: { collectionId: 'theme', mode: 'Dark', name: 'Night' } },
    variant === 'two-collections' ? { method: 'collection.renameMode', params: { collectionId: 'other', mode: 'Dark', name: 'Night' } } :
      variant === 'two-modes' ? { method: 'collection.renameMode', params: { collectionId: 'theme', mode: 'Light', name: 'Day' } } :
        { method: 'variable.setValue', params: { variableId: 'v0', mode: 'Night', value: 95 } }]
  let result: ReturnType<typeof api.execute>
  act(() => { result = api.execute({ method: 'batch.apply', params: { operations }, documentId: doc.id, session: e.sessionGeneration, expectedRevision: version }) })
  if (!result!.ok) {
    expect(e.state.doc).toBe(before); expect(e.state.version).toBe(version); expect(e.canUndo).toBe(false)
    expect(result!.error.code + ' ' + result!.error.message).toMatch(/unsupported|separate|split|不支持|单独|分开/i)
    return
  }
  expect(e.state.doc.nodes[owner].instance!.overrides).toEqual(before.nodes[owner].instance!.overrides)
  expect(originNode(e.state.doc, owner, childKey).width).toBe(variant === 'value-edit' ? 95 : 80)
  act(() => e.change('Later source mode after batch', d => { d.nodes[layer.id].variableModes = { theme: variant === 'two-modes' ? 'Day' : variant === 'name-swap' ? 'Dark' : 'Light' } }))
  expect(originNode(e.state.doc, owner, childKey).width).toBe(10)
  expect(e.state.doc.nodes[owner].instance!.overrides).toEqual(before.nodes[owner].instance!.overrides)
})
